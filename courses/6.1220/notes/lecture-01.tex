\begin{problem}{Interval Scheduling}
Consider a set of $n$ requests $R = \{(a_i, b_i)\}_{i = 1}^n$, where:
\begin{itemize}
    \item Every \emph{request} $r_i := (a_i, b_i)$ is an open interval with real number endpoints $a_i< b_i$.
    \item Two requests $r_i, r_j$ are \emph{compatible} if they do not intersect.
\end{itemize}
Determine any largest possible subset of mutually compatible requests.
\end{problem}

\textbf{Example.} If $R = \{(1, 2), (1, 3), (1, 4)\}$, then any one of $\{(1, 2)\}$, $\{(1, 3)\}$, or $\{(1, 4)\}$ is an optimal solution.

\begin{solution}{Interval Scheduling}
Greedily pick the compatible requests that end earliest.
\end{solution}

\begin{proof}
We define the following:
\begin{itemize}
    \item Let $S_{\textsc{GREEDY}} := \{s_i\}_{i = 1}^m$ be the \emph{ordered} greedy solution (sorted by endpoint).
    \item Let $S_{\textsc{OPT}} := \{s_i'\}_{i = 1}^{m'}$ be the \emph{ordered} optimal solution (sorted by endpoint).
\end{itemize}
If $S_{\textsc{GREEDY}} = S_{\textsc{OPT}}$, then we win.  Otherwise, let $j^*$ be the first index where $S_{\textsc{GREEDY}}$ and $S_{\textsc{OPT}}$ differ.
\begin{quote}
    \textbf{Claim 1.} If $j^* > 1$, then $b_{j^* - 1} \leq a_{j^*}'$.  This is because $s_{j^* - 1} = s_{j^* - 1}'$ is assumed compatible with $s_{j^*}'$.

    \textbf{Claim 2.} We must have $b_{j^*} \leq b_{j^*}'$.  This is because the greedy algorithm picked $s_{j^*}$, not $s_{j^*}'$.

    \textbf{Claim 3.} By Claim 1 and Claim 2, the optimal solution can be adjusted like so:
    \[ \{s_1, \dots, s_{j^* - 1}, s_{j^*}', \dots, s_{m'}'\} \mapsto \{s_1, \dots, s_{j^*}, s_{j^* + 1}', \dots, s_{m'}'\}. \]
\end{quote}
By repeatedly applying Claim 3, we see $S = \{s_1, \dots, s_m, s_{m + 1}', \dots, s_{m'}'\}$ is an optimal solution.  But $m' > m$ is impossible!  If it were, the greedy algorithm would have added $(s_{m + 1}', \dots, s_{m'}')$, yet it didn't.  So $m = m'$. ~ $\blacksquare$
\end{proof}

\begin{problem}{Weighted Interval Scheduling}
Consider a set of $n$ requests $R = \{(a_i, b_i)\}_{i = 1}^n$ together with weights $\{w_i\}_{i = 1}^n$.  Determine any \emph{heaviest} possible subset of compatible \emph{weighted} requests.
\end{problem}

\begin{solution}{Weighted Interval Scheduling}
    Just use DP.  Sort the intervals by start time.  Then:
    \[ W(R) = \max \{ W(\{r_2, \dots, r_n\}), \ \  w_1 + W(\{r_j \mid j \geq t_1\})\}, \]
    where $t_i$ is the least index $j > i$ for which $r_j$ does not intersect $r_i$.  Repeat, recurse, blah blah.
\end{solution}

\begin{remark}
    For any $i$, how can you find $t_i$ in $O(\log n)$ time?
    
    How can you find all the $t_i$ in $O(n)$ time, given the ordering of all interval endpoints?
\end{remark}

\begin{problem}{Online Interval Scheduling}
Determine any largest possible subset of mutually compatible requests, given that requests come one at a time (in \emph{any} order!) and must be accepted or rejected on arrival.
\end{problem}
