\subsection{Problem Types}

We need to distinguish \textbf{problems} from \textbf{instances.}
\begin{itemize}
    \item The minimum cut problem asks for a minimal cut in any instance of a graph $G$.
    \item The shortest path problem asks for a shortest path on any instance of a graph $G$.
\end{itemize}
The \textbf{size} of an instance is the number of bits needed to encode it.
\begin{itemize}
    \item The size of a graph is $|V| + |E|$, roughly.
    \item But actually, it's more like $|V| + 3|E|$ because each edge needs labels\dots
    \item But actually, it's more complicated because it depends on the size of the edge weights, and so on\dots
\end{itemize}
We say an algorithm runs in time $O(t(n))$ if it can solve any instance of size $n$ in time $O(t(n))$.

\begin{remark}
    Instead of \emph{time}, we could also consider \emph{space}, or \emph{in-place}-ness, or \emph{randomness} as limited resources\dots
\end{remark}

An algorithm is \textbf{tractable} if has a runtime of $O(n^k)$ for some constant $k$, and \textbf{intractable} otherwise.
\begin{itemize}
    \item From lecture, we know MST, min-cut, and shortest-path are all tractable.
    \item But max-cut and longest-path are not intractable.
    \begin{itemize}
        \item Max-cut can be approximated easily, though!  But not longest-path\dots
    \end{itemize}
\end{itemize}
There are many variants of problems.
\begin{itemize}
    \item \textbf{Decision:} Instances have YES / NO solutions.
    \item \textbf{Search:} Instances have solutions of a certain quality of weight.
    \item \textbf{Optimization:} Instances have solutions of a maximal quality or weight.
\end{itemize}
So the MST problem can have the following variants:
\begin{itemize}
    \item \textbf{Decision:} Does $G$ have a spanning tree of weight at most $K$?
    \item \textbf{Search:} Find a spanning tree of $G$ of weight at most $K$, if it exists.
    \item \textbf{Optimization:} Find a minimum spanning tree of $G$.
\end{itemize}

\subsection{P, NP, coNP, EXP, and R}

Only consider problems $\pi$ that are decision variants.

\textbf{Definition (P).} A problem $\pi$ is solvable in (deterministic) \textbf{polynomial} time iff there exists a (deterministic) algorithm $A_{\pi}$ and a constant $k$ such that for every instance $x$, we have:
\[ A_{\pi}(x) \text{ runs in time } O(|x|^c). ~~~ \textbf{ AND } ~~~ A_{\pi}(x) = \pi(x). \]

\textbf{Definition (NP).} A problem $\pi$ is solvable in \textbf{non-deterministic polynomial} time iff there exists a (deterministic) verifier algorithm $V_{\pi}$ and constants $c, c'$ such that for every instance $x$ of size $|x| = n$, we have:
\begin{align*}
& V_{\pi}(x, y) \text{ runs in time } O((|x| + |y|)^c). \\
\textbf{ AND } ~~~ & \pi(x) = \text{YES} \text{ iff there is a } y \text{ with } |y| \leq n^{c'} \text{ and } V_{\pi}(x, y) = \text{YES}. \end{align*}
All of min-cut, max-cut, min-path, and max-path are \textbf{NP}, with the desired $y$ being the cut / path itself.

\begin{remark}
    Reminder that we're considering the decision variants of min-cut, max-cut, min-path, and max-path.
\end{remark}

\textbf{Definition (coNP).} A problem $\pi$ is solvable in \textbf{co-non-deterministic polynomial} time iff there exists a (deterministic) verifier algorithm $V_{\pi}$ and constants $c, c'$ such that for every instance $x$ of size $|x| = n$, we have:
\begin{align*}
& V_{\pi}(x, y) \text{ runs in time } O((|x| + |y|)^c). \\
\textbf{ AND } ~~~ & \pi(x) = \text{NO} \text{ iff there is a } y \text{ with } |y| \leq n^{c'} \text{ and } V_{\pi}(x, y) = \text{YES}. \end{align*}
It's easy to find \textbf{coNP} problems by negating \textbf{NP} problems; ``are all paths of weight $\leq k$?'', for example.

\begin{claim}
    If $\pi \in \textbf{P}$, then $\pi \in \textbf{NP}$ and $\pi \in \textbf{coNP}$.
\end{claim}

\begin{proof}
    The desired \textbf{NP} and \textbf{coNP} algorithms just run the \textbf{P} algorithm. ~ $\blacksquare$
\end{proof}

These are the possible relationships between \textbf{P}, \textbf{NP}, and \textbf{coNP}. \vspace{0.3cm}

\begin{claim}
    It is not possible that $\textbf{P} = \textbf{NP} \neq \textbf{coNP}$.
\end{claim}

\begin{proof}
    The intuition is that if a problem is in \textbf{P}, then \textbf{NP} and \textbf{coNP} are equally easy.

    Formally, take any $\pi \in \textbf{coNP}$.  Then $\overline{\pi} \in \textbf{NP} = \textbf{P} \subseteq \textbf{coNP}$, so $\overline{\pi} \in \textbf{coNP}$.  But then $\pi = \overline{\overline{\pi}} \in \textbf{NP} = \textbf{P}$, done. ~ $\blacksquare$
\end{proof}

\textbf{Definition (EXP).} A problem $\pi$ is solvable in (deterministic) \textbf{polynomial} time iff there exists a (deterministic) algorithm $A_{\pi}$ and a constant $k$ such that for every instance $x$, we have:
\[ A_{\pi}(x) \text{ runs in time } O(2^{|x|^c}). ~~~ \textbf{ AND } ~~~ A_{\pi}(x) = \pi(x). \]

\begin{claim}
    All of \textbf{P}, \textbf{NP}, and \textbf{coNP} are in \textbf{EXP}.
\end{claim}

\begin{proof}
    Obviously $\textbf{P} \subseteq \textbf{EXP}$ because polynomial-time is faster than exponential-time.

    For any $\pi \in \textbf{NP}$, we can solve $\pi$ by checking all $2^{\left(n^{c'}\right)}$ possible certificates. ~ $\blacksquare$
\end{proof}

\begin{remark}
    We know that $\textbf{P} \neq \textbf{EXP}$.  This is the Deterministic Time Hierarchy Theorem.

    The reason why we care about \textbf{EXP} is because it's a ``higher level'' of difficulty than \textbf{NP} and \textbf{coNP}.
\end{remark}

\textbf{Definition (R).} A problem $\pi$ is \textbf{solvable} iff there exists a (deterministic) algorithm $A_{\pi}$ such that for every instance $x$, we have:
\[ A_{\pi}(x) \text{ runs in finite time.} ~~~ \textbf{ AND } ~~~ A_{\pi}(x) = \pi(x). \]
